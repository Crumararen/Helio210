     output_buffer[0x10000] = []    //                32 kilobytes

FOR EACH line IN source_file:
    tokens = Tokenize(line)
    IF tokens is empty OR tokens[0] is a comment: CONTINUE

    // --- Directives ---
    IF tokens[0] == ".ORG":
        Set_Org_State(Parse_Number(tokens[1]))
        CONTINUE
    IF tokens[0] == ".BYTE":
        FOR EACH t IN tokens[1..]:
            Emit_Byte(Resolve_Value_Or_Label(t, 0xFF, output_offset, 1, 0xFF))
        CONTINUE
    IF tokens[0] starts with '.':
        PrintError("Unknown directive: " + tokens[0])
        error_flag = TRUE
        CONTINUE

    // --- Label definition: "name:" ---
    IF tokens[0] ends with ':':
        name = Strip(tokens[0], ':')
        IF Is_Valid_Label_Name(name) == FALSE:
            PrintError("Invalid label name: " + name)
            error_flag = TRUE
            CONTINUE
        IF Find_Label(name) != NOT_FOUND:
            PrintError("Duplicate label: " + name)
            error_flag = TRUE
            CONTINUE
        Define_Label(name, Calculate_Current_Address())
        tokens = tokens[1..]
        IF tokens is empty: CONTINUE

    // --- Instruction ---
    mnemonic = tokens[0]
    ops      = tokens[1..]

    instr = Lookup_By_Mnemonic(mnemonic)
    IF instr == NOT_FOUND:
        PrintError("Unknown mnemonic: " + mnemonic)
        error_flag = TRUE
        CONTINUE

    IF Count(ops) != instr.ops:
        PrintError(mnemonic + " expects " + instr.ops + " operand(s), got " + Count(ops))
        error_flag = TRUE
        CONTINUE

    Encode_Instruction(instr, ops)

FUNCTION Encode_Instruction(instr, ops):
    SWITCH instr.format:

        // 24-bit immediate, first byte carries a register
        CASE "imm24_reg":                       // LOADI Rn, #imm  /  JZ Rn, target
            ra = Parse_Register(ops[0])
            Emit_Byte(instr.opcode | ra)
            imm = Resolve_Value_Or_Label(ops[1], 0xFFFFFF, output_offset, 3, 0xFFFFFF)
            Emit_Byte((imm >> 16) & 0xFF)
            Emit_Byte((imm >>  8) & 0xFF)
            Emit_Byte( imm        & 0xFF)

        // 24-bit immediate, no register
        CASE "imm24":                           // JMP target  /  CALL target
            ASSERT (instr.opcode & 0x0F) == 0   // low nibble reserved
            Emit_Byte(instr.opcode)
            imm = Resolve_Value_Or_Label(ops[0], 0xFFFFFF, output_offset, 3, 0xFFFFFF)
            Emit_Byte((imm >> 16) & 0xFF)
            Emit_Byte((imm >>  8) & 0xFF)
            Emit_Byte( imm        & 0xFF)

        // Three registers
        CASE "reg3":                            // JLT, JGT, WCHR, RCHR
            ra = Parse_Register(ops[0])
            rb = Parse_Register(ops[1])
            rc = Parse_Register(ops[2])
            Emit_Byte(instr.opcode | ra)
            Emit_Byte((rb << 4) | rc)

        // One register
        CASE "reg1":                            // PUSH, POP
            ra = Parse_Register(ops[0])
            Emit_Byte(instr.opcode | ra)
            Emit_Byte(0x00)

        // No registers
        CASE "reg0":                            // RET
            Emit_Byte(instr.opcode)
            Emit_Byte(0x00)

        // Shift: opcode|R, (T<<7)|(value & 0x3F)
        CASE "shift":                           // SHL, SHR
            ra = Parse_Register(ops[0])
            Emit_Byte(instr.opcode | ra)
            IF Is_Register(ops[1]):
                t   = 1
                val = Parse_Register(ops[1])
            ELSE:
                t   = 0
                val = Resolve_Value_Or_Label(ops[1], 0x3F, output_offset, 1, 0x3F)
            Emit_Byte((t << 7) | (val & 0x3F))

        // RND: opcode|R, imm3
        CASE "imm3":                            // RND Rn, #imm3
            ra = Parse_Register(ops[0])
            Emit_Byte(instr.opcode | ra)
            imm = Resolve_Value_Or_Label(ops[1], 0x07, output_offset, 1, 0x07)
            Emit_Byte(imm & 0x07)

        // Sub-coded
        CASE "sub_2reg":                        // LOAD, STORE, ADD, ..., MOV
            ra = Parse_Register(ops[0])
            rb = Parse_Register(ops[1])
            Emit_Byte(instr.opcode)             // full byte: (0<<4)|subcode
            Emit_Byte((ra << 4) | rb)

        CASE "sub_1reg":                        // NOT, MOVPC
            ra = Parse_Register(ops[0])
            Emit_Byte(instr.opcode)
            Emit_Byte(ra << 4)

        CASE "sub_jmpr":                        // JMPR (flag 0x0), CALLR (flag 0xF)
            ra = Parse_Register(ops[0])
            Emit_Byte(instr.opcode)
            Emit_Byte((ra << 4) | instr.flag)

        CASE "sub_0reg":                        // NOP, HALT
            Emit_Byte(instr.opcode)
            Emit_Byte(0x00)

        DEFAULT:
            PrintError("Unhandled instruction format: " + instr.format)
            error_flag = TRUE


FUNCTION Set_Org_State(addr):
    FOR EACH region IN REGIONS:
        IF addr >= region.start AND addr <= region.end:
            IF region.is_protected:
                PrintError("Cannot assemble into protected region: " + region.name)
                error_flag = TRUE
                RETURN
            current_org    = addr
            current_region = region
            region_end     = region.end
            output_offset  = addr           // assume .ORG resets the output pointer
            RETURN
    PrintError(".ORG address not in a valid region: " + addr)
    error_flag = TRUE

FUNCTION Calculate_Current_Address():
    // Address of the next byte to be emitted.
    // Note: assumes .ORG rebases output_offset so current_org + output_offset
    // is only correct if .ORG resets output_offset. Preserved from original.
    RETURN output_offset

FUNCTION Resolve_Value_Or_Label(arg_string, max_value, fixup_offset, fixup_bytes, fixup_mask):
    // 1. Already-defined label
    label_addr = Find_Label(arg_string)
    IF label_addr != NOT_FOUND:
        IF label_addr > max_value:
            PrintError("Symbol " + arg_string + " (=" + label_addr + ") exceeds field limit " + max_value)
            error_flag = TRUE
            RETURN label_addr & max_value
        RETURN label_addr

    // 2. Char literal
    IF arg_string starts with "'" AND arg_string ends with "'":
        ch = Parse_Char_Literal(arg_string)
        IF ch > max_value:
            PrintError("Char literal exceeds field limit")
            error_flag = TRUE
            RETURN ch & max_value
        RETURN ch

    // 3. Numeric literal
    IF Is_Numeric_Literal(arg_string):
        value = Parse_Number(arg_string)
        IF value > max_value:
            PrintError("Value " + value + " exceeds limit " + max_value)
            error_flag = TRUE
            value = value & max_value
        RETURN value

    // 4. Forward reference — queue a fixup (patch output_buffer later)
    IF fixup_bytes == 0:
        PrintError("Undefined symbol: " + arg_string)
        error_flag = TRUE
        RETURN 0
    IF fixup_count >= MAX_FIXUPS:
        PrintError("Too many fixups (max " + MAX_FIXUPS + ")")
        error_flag = TRUE
        RETURN 0
    fixups[fixup_count] = { offset: fixup_offset, label: arg_string,
                            num_bytes: fixup_bytes, mask: fixup_mask }
    fixup_count += 1
    RETURN 0

FUNCTION Emit_Byte(val):
    addr = Calculate_Current_Address()
    IF addr > region_end:
        PrintError("Output exceeds end of region " + current_region.name)
        error_flag = TRUE
        RETURN
    IF output_offset >= OUTPUT_SIZE:
        PrintError("Output buffer overflow")
        error_flag = TRUE
        RETURN
    output_buffer[output_offset] = val & 0xFF
    output_offset += 1

FUNCTION Parse_Register(string):
    IF Length(string) < 2 OR (string[0] != 'R' AND string[0] != 'r'):
        PrintError("Invalid register: " + string)
        error_flag = TRUE
        RETURN 0
    num = Parse_Number(string[1:])
    IF num < 0 OR num > 15:
        PrintError("Invalid register (must be R0-R15): " + string)
        error_flag = TRUE
        RETURN 0
    RETURN num

FUNCTION Find_Label(name):
    FOR i = 0 TO label_count - 1:
        IF labels[i].name == name: RETURN labels[i].addr
    RETURN NOT_FOUND

FUNCTION Define_Label(name, addr):
    IF label_count >= MAX_LABELS:
        PrintError("Too many labels (max " + MAX_LABELS + ")")
        error_flag = TRUE
        RETURN
    labels[label_count] = { name: name, addr: addr }
    label_count += 1

FUNCTION Apply_Fixups():
    FOR i = 0 TO fixup_count - 1:
        f = fixups[i]
        addr_val = Find_Label(f.label)
        IF addr_val == NOT_FOUND:
            PrintError("Undefined symbol: " + f.label)
            error_flag = TRUE
            CONTINUE
        IF addr_val > f.mask:
            PrintError("Symbol " + f.label + " (=" + addr_val + ") exceeds field limit " + f.mask)
            error_flag = TRUE
            CONTINUE
        FOR b = 0 TO f.num_bytes - 1:
            shift = (f.num_bytes - 1 - b) * 8
            output_buffer[f.offset + b] |= (addr_val >> shift) & 0xFF

// ASSUMED EXTERNAL HELPERS

FUNCTION Is_Numeric_Literal(s):
    IF Length(s) == 0: RETURN FALSE
    IF Length(s) >= 3 AND s[0] == "'" AND s[Length(s)-1] == "'": RETURN TRUE
    IF Length(s) > 2 AND (s[0..1] == "0x" OR s[0..1] == "0X"): RETURN Is_All_Hex(s[2:])
    IF Length(s) > 2 AND (s[0..1] == "0b" OR s[0..1] == "0B"): RETURN Is_All_Bin(s[2:])
    RETURN Is_All_Digits(s)

FUNCTION Is_Register(s):
    IF Length(s) < 2: RETURN FALSE
    IF s[0] != 'R' AND s[0] != 'r': RETURN FALSE
    n = Parse_Number(s[1:])
    RETURN n >= 0 AND n <= 15
